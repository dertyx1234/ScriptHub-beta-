-- Helper Function: Create Elements
local function CreateToggle(parent, text, settingName, yPos)
 local Container = Instance.new("Frame")
 Container.Size = UDim2.new(1, -30, 0, 35)
 Container.Position = UDim2.new(0, 15, 0, yPos)
 Container.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
 Container.Parent = parent
 
 local Corner = Instance.new("UICorner")
 Corner.CornerRadius = UDim.new(0, 5)
 Corner.Parent = Container
 
 local Label = Instance.new("TextLabel")
 Label.Size = UDim2.new(0.7, 0, 1, 0)
 Label.BackgroundTransparency = 1
 Label.Text = text
 Label.TextColor3 = Color3.fromRGB(255, 255, 255)
 Label.Font = Enum.Font.Gotham
 Label.TextSize = 14
 Label.TextXAlignment = Enum.TextXAlignment.Left
 Label.Parent = Container
 
 local Toggle = Instance.new("TextButton")
 Toggle.Size = UDim2.new(0.3, 0, 1, 0)
 Toggle.Position = UDim2.new(0.7, 0, 0, 0)
 Toggle.BackgroundColor3 = Color3.fromRGB(60, 60, 60)
 Toggle.Text = "OFF"
 Toggle.TextColor3 = Color3.fromRGB(200, 200, 200)
 Toggle.Font = Enum.Font.GothamBold
 Toggle.TextSize = 12
 Toggle.Parent = Container
 
 local IsOn = false
 Toggle.MouseButton1Click:Connect(function()
 IsOn = not IsOn
 Toggle.BackgroundColor3 = IsOn and Color3.fromRGB(0, 255, 128) or Color3.fromRGB(60, 60, 60)
 Toggle.Text = IsOn and "ON" or "OFF"
 print(settingName.. " set to ".. tostring(IsOn))
 end)
 
 return Toggle
end

local function CreateSlider(parent, text, minVal, maxVal, currentVal, yPos)
 local Container = Instance.new("Frame")
 Container.Size = UDim2.new(1, -30, 0, 40)
 Container.Position = UDim2.new(0, 15, 0, yPos)
 Container.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
 Container.Parent = parent
 
 local Corner = Instance.new("UICorner")
 Corner.CornerRadius = UDim.new(0, 5)
 Corner.Parent = Container
 
 local Label = Instance.new("TextLabel")
 Label.Size = UDim2.new(1, 0, 0, 20)
 Label.BackgroundTransparency = 1
 Label.Text = text.. " [".. tostring(currentVal).. "]"
 Label.TextColor3 = Color3.fromRGB(255, 255, 255)
 Label.Font = Enum.Font.Gotham
 Label.TextSize = 13
 Label.TextXAlignment = Enum.TextXAlignment.Left
 Label.Parent = Container
 
 local SliderBar = Instance.new("Frame")
 SliderBar.Size = UDim2.new(1, -40, 0, 10)
 SliderBar.Position = UDim2.new(0, 20, 0, 30)
 SliderBar.BackgroundColor3 = Color3.fromRGB(20, 20, 20)
 SliderBar.Parent = Container
 
 local SliderCorner = Instance.new("UICorner")
 SliderCorner.CornerRadius = UDim.new(0, 5)
 SliderCorner.Parent = SliderBar
 
 local SliderFill = Instance.new("Frame")
 SliderFill.Size = UDim2.new(0, (currentVal - minVal) / (maxVal - minVal), 1, 0)
 SliderFill.BackgroundColor3 = Color3.fromRGB(0, 255, 128)
 SliderFill.Parent = SliderBar
 
 local Handle = Instance.new("TextButton")
 Handle.Size = UDim2.new(0, 20, 0, 20)
 Handle.Position = UDim2.new(0, ((currentVal - minVal) / (maxVal - minVal)) * 100 - 10, 0, -5)
 Handle.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
 Handle.Text = ""
 Handle.Parent = Container
 
 local function UpdateSlider(Value)
 local Percent = (Value - minVal) / (maxVal - minVal)
 SliderFill.Size = UDim2.new(Percent, 0, 1, 0)
 Handle.Position = UDim2.new(Percent, -10, 0, -5)
 Label.Text = text.. " [".. tostring(math.floor(Value)).. "]"
 end
 
 local isDragging = false
 Handle.InputBegan:Connect(function(input)
 if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
 isDragging = true
 end
 end)
 
 Handle.InputEnded:Connect(function(input)
 if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
 isDragging = false
 end
 end)
 
 UserInputService.InputChanged:Connect(function(input)
 if isDragging then
 local pos = Handle.AbsolutePosition.X
 local width = SliderBar.AbsoluteSize.X
 local percent = math.clamp(pos / width, 0, 1)
 local newVal = minVal + (percent * (maxVal - minVal))
 UpdateSlider(newVal)
 -- Logic for actual value change would go here
 end
 end)
end

-- Tab Switching Logic

 TabButtons[i].MouseButton1Click:Connect(function()
 -- Update Button Colors
 for j, Btn in ipairs(TabButtons) do
 Btn.BackgroundColor3 = Tabs[j].Active and Color3.fromRGB(0, 255, 128) or Color3.fromRGB(50, 50, 50)
 end
 
 -- Update Tab Visibility
 for j, Frame in ipairs(TabFrames) do

 Frame.Visible = true

 Frame.Visible = false

 end
 end)
