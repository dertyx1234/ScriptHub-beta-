-- Dragging
local dragging = false
local dragInput = nil
local mouseStart = nil
local frameStart = nil

titleBar.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragInput = input
        mouseStart = input.Position
        frameStart = frame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
                dragInput = nil
            end
        end)
    end
end)

titleBar.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - mouseStart
        frame.Position = UDim2.new(frameStart.X.Scale, frameStart.X.Offset + delta.X, frameStart.Y.Scale, frameStart.Y.Offset + delta.Y)
    end
end)

-- UI actions
toggle.MouseButton1Click:Connect(function()
    frame.Visible = not frame.Visible
end)

closeBtn.MouseButton1Click:Connect(function()
    gui:Destroy()
    flying = false
    movingUp = false
    movingDown = false
end)

flyBtn.MouseButton1Click:Connect(function()
    flying = not flying
    if flying then
        flyBtn.Text = "FLIGHT: ON"
        flyBtn.BackgroundColor3 = Color3.fromRGB(0, 170, 255)
        fbs.Color = Color3.fromRGB(0, 210, 255)
        status.Text = "Status: Online"
        status.TextColor3 = Color3.fromRGB(100, 255, 150)
    else
        flyBtn.Text = "FLIGHT: OFF"
        flyBtn.BackgroundColor3 = Color3.fromRGB(25, 25, 35)
        fbs.Color = Color3.fromRGB(100, 100, 255)
        status.Text = "Status: Offline"
        status.TextColor3 = Color3.fromRGB(160, 160, 170)
    end
end)

spdMinus.MouseButton1Click:Connect(function()
    speed = math.max(20, speed - 10)
    speedLabel.Text = "Speed: " .. speed
end)

spdPlus.MouseButton1Click:Connect(function()
    speed = math.min(200, speed + 10)
    speedLabel.Text = "Speed: " .. speed
end)

upBtn.MouseButton1Down:Connect(function() movingUp = true end)
upBtn.MouseButton1Up:Connect(function() movingUp = false end)
downBtn.MouseButton1Down:Connect(function() movingDown = true end)
downBtn.MouseButton1Up:Connect(function() movingDown = false end)

lp.CharacterAdded:Connect(function()
    flying = false
    movingUp = false
    movingDown = false
    flyBtn.Text = "FLIGHT: OFF"
    flyBtn.BackgroundColor3 = Color3.fromRGB(25, 25, 35)
    fbs.Color = Color3.fromRGB(100, 100, 255)
    status.Text = "Status: Offline"
    status.TextColor3 = Color3.fromRGB(160, 160, 170)
end)

-- Flight loop
RunService.RenderStepped:Connect(function()
    if not flying then return end
    local c = lp.Character
    if not c then return end
    local hrp = c:FindFirstChild("HumanoidRootPart")
    local hum = c:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum then return end

    hum.PlatformStand = true

    local move = hum.MoveDirection
    if move.Magnitude > 0 then move = move.Unit else move = Vector3.zero end

    local horizVel = move * speed

    local vertVel = 0
    if movingUp then vertVel = speed end
    if movingDown then vertVel = -speed end
    if movingUp and movingDown then vertVel = 0 end

    hrp.AssemblyLinearVelocity = Vector3.new(horizVel.X, vertVel, horizVel.Z)
end)
